import { EdgeInfer } from '@edgeinfer/core';

// Initialize WebGPU context with fallback
const engine = new EdgeInfer({
    backend: 'webgpu',
    wasmFallback: true,
    cacheModels: true
});

// Load quantized ONNX model
await engine.loadModel(
    'tinybert-sentiment-int8', 
    'https://models.edgeinfer.ai/tinybert.onnx'
);

console.log('Neural engine ready.');
const text = "This library is absolutely mind-blowing!";

// Zero-copy tokenization in WASM
const tokens = await engine.tokenize(text);

// Execute inference graph on GPU
const start = performance.now();
const result = await engine.run(tokens);
const latency = performance.now() - start;

console.log({
    sentiment: result.label,      // "POSITIVE"
    confidence: result.score,     // 0.998
    latencyMs: latency            // ~4.2ms
});
